# Cleans up your X Feed to only show accounts you follow and nothing else  

``` javascript
// ==UserScript==
// @name         X Follow-Only Feed Filter
// @namespace    jack.x.followfilter
// @version      1.1.0
// @description  Hides X home-feed tweets from accounts X confirms you do not follow.
// @match        https://x.com/*
// @match        https://twitter.com/*
// @run-at       document-idle
// @grant        none
// ==/UserScript==

(() => {
    'use strict';


    // ============================================================
    // CONFIG
    // ============================================================

    const DEBUG = true;

    /*
     * Known relationship remains valid for 24 hours.
     */
    const CACHE_TTL =
        24 * 60 * 60 * 1000;

    /*
     * If we cannot determine the relationship, retry later.
     */
    const UNKNOWN_TTL =
        5 * 60 * 1000;

    /*
     * Maximum time to wait for X's More menu.
     */
    const MENU_TIMEOUT =
        2500;

    /*
     * Delay between account checks so we don't machine-gun
     * X's menus while scrolling.
     */
    const CHECK_DELAY =
        500;


    // ============================================================
    // CONSTANTS
    // ============================================================

    /*
     * New key intentionally ignores the bad UNKNOWN cache
     * generated by v1.0's iframe experiment.
     */
    const CACHE_KEY =
        'x-follow-only-filter-cache-v2';

    const HIDDEN_ATTR =
        'data-x-follow-filter-hidden';

    const HANDLE_ATTR =
        'data-x-follow-filter-handle';


    const REL = {
        FOLLOWING: 'following',
        NOT_FOLLOWING: 'not-following',
        UNKNOWN: 'unknown',
        BUSY: 'busy'
    };


    // ============================================================
    // STATE
    // ============================================================

    let cache =
        loadCache();

    const queuedHandles =
        new Set();

    const checkQueue =
        [];

    const pendingRoots =
        new Set();

    let workerRunning =
        false;

    let scanScheduled =
        false;

    let blockedCount =
        0;


    // ============================================================
    // LOG
    // ============================================================

    console.info(
        '%c[X Follow Filter] v1.1 LOADED',
        [
            'background:#111',
            'color:#00ff88',
            'font-weight:bold',
            'padding:4px 7px'
        ].join(';'),
        location.href
    );


    function log(...args) {

        if (!DEBUG) {
            return;
        }

        console.log(
            '%c[X Follow Filter]',
            [
                'background:#111',
                'color:#1d9bf0',
                'font-weight:bold',
                'padding:2px 5px'
            ].join(';'),
            ...args
        );
    }


    // ============================================================
    // CSS
    // ============================================================

    function installCSS() {

        const style =
            document.createElement('style');

        style.textContent = `
            article[data-testid="tweet"][${HIDDEN_ATTR}="1"] {
                display: none !important;
            }
        `;

        (
            document.head ||
            document.documentElement
        ).appendChild(style);
    }


    // ============================================================
    // HELPERS
    // ============================================================

    function sleep(ms) {

        return new Promise(
            resolve =>
                setTimeout(resolve, ms)
        );
    }


    function normalizeText(element) {

        return (
            element?.textContent || ''
        )
            .replace(
                /[\u200B-\u200D\u2060\uFEFF]/g,
                ''
            )
            .replace(
                /\s+/g,
                ' '
            )
            .trim();
    }


    function isHome() {

        return (
            location.pathname === '/home' ||
            location.pathname === '/'
        );
    }


    // ============================================================
    // CACHE
    // ============================================================

    function loadCache() {

        try {

            return JSON.parse(
                localStorage.getItem(
                    CACHE_KEY
                ) || '{}'
            );

        } catch {

            return {};
        }
    }


    function saveCache() {

        try {

            localStorage.setItem(
                CACHE_KEY,
                JSON.stringify(cache)
            );

        } catch (error) {

            log(
                'Could not save cache',
                error
            );
        }
    }


    function cacheRelationship(
        handle,
        relationship
    ) {

        cache[
            handle.toLowerCase()
        ] = {

            handle,

            relationship,

            checkedAt:
                Date.now()
        };


        saveCache();
    }


    function getCachedRelationship(
        handle
    ) {

        const key =
            handle.toLowerCase();

        const entry =
            cache[key];


        if (!entry) {
            return null;
        }


        const age =
            Date.now() -
            entry.checkedAt;


        const ttl =
            entry.relationship ===
                REL.UNKNOWN

                ? UNKNOWN_TTL
                : CACHE_TTL;


        if (age > ttl) {

            delete cache[key];

            saveCache();

            return null;
        }


        return entry.relationship;
    }


    // ============================================================
    // TWEET AUTHOR
    // ============================================================

    function getTweetHandle(tweet) {

        if (!(tweet instanceof Element)) {
            return null;
        }


        const username =
            tweet.querySelector(
                '[data-testid="User-Name"]'
            );


        if (!username) {
            return null;
        }


        /*
         * Prefer profile links like:
         *
         *     /Reuters
         *
         * and reject:
         *
         *     /Reuters/status/12345
         */
        for (
            const link
            of username.querySelectorAll(
                'a[href]'
            )
        ) {

            const href =
                link.getAttribute(
                    'href'
                );


            const match =
                href?.match(
                    /^\/([A-Za-z0-9_]{1,15})\/?$/
                );


            if (match) {
                return match[1];
            }
        }


        /*
         * Text fallback.
         */
        const match =
            normalizeText(username)
                .match(
                    /@([A-Za-z0-9_]{1,15})/
                );


        return (
            match
                ? match[1]
                : null
        );
    }


    // ============================================================
    // FIND OUTER TWEET'S MORE BUTTON
    // ============================================================

    function getTweetCaret(tweet) {

        const candidates =
            tweet.querySelectorAll(
                'button[data-testid="caret"]'
            );


        for (
            const button
            of candidates
        ) {

            /*
             * Prevent a quoted/nested tweet's menu from being
             * mistaken for the outer tweet's menu.
             */
            if (
                button.closest(
                    'article[data-testid="tweet"]'
                ) === tweet
            ) {

                return button;
            }
        }


        return null;
    }


    // ============================================================
    // MENU UTILITIES
    // ============================================================

    function isVisible(element) {

        if (!(element instanceof Element)) {
            return false;
        }


        const rect =
            element.getBoundingClientRect();


        return (
            rect.width > 0 &&
            rect.height > 0
        );
    }


    function getVisibleMenus() {

        return [
            ...document.querySelectorAll(
                '[role="menu"]'
            )
        ].filter(
            isVisible
        );
    }


    /*
     * Do not interfere if the USER already has a menu open.
     */
    function userMenuIsOpen() {

        return (
            getVisibleMenus().length > 0
        );
    }


    function getMenuItemTexts(menu) {

        const elements =
            menu.querySelectorAll(
                [
                    '[role="menuitem"]',
                    '[role="button"]',
                    'button'
                ].join(',')
            );


        const texts =
            [];


        for (
            const element
            of elements
        ) {

            const text =
                normalizeText(
                    element
                );


            if (
                text &&
                !texts.includes(text)
            ) {

                texts.push(text);
            }
        }


        /*
         * Fallback if X changes the role structure.
         */
        if (texts.length === 0) {

            const whole =
                normalizeText(
                    menu
                );


            if (whole) {
                texts.push(whole);
            }
        }


        return texts;
    }


    // ============================================================
    // RELATIONSHIP DETECTION FROM MORE MENU
    // ============================================================

    function classifyMenu(
        texts,
        handle
    ) {

        const normalizedHandle =
            handle.toLowerCase();


        /*
         * Look at every individual menu command.
         */
        for (
            const raw
            of texts
        ) {

            const text =
                raw
                    .replace(/\s+/g, ' ')
                    .trim();


            const lower =
                text.toLowerCase();


            /*
             * Examples we expect:
             *
             *     Unfollow @Reuters
             *     Unfollow Reuters
             *     Unfollow
             */
            if (
                lower === 'unfollow'
            ) {

                return REL.FOLLOWING;
            }


            if (
                lower.startsWith(
                    'unfollow '
                )
            ) {

                /*
                 * If X includes a username, make sure it
                 * corresponds to this tweet.
                 */
                const possibleHandle =
                    lower
                        .slice(
                            'unfollow '.length
                        )
                        .replace(/^@/, '')
                        .trim();


                if (
                    !possibleHandle ||
                    possibleHandle ===
                        normalizedHandle
                ) {

                    return REL.FOLLOWING;
                }


                /*
                 * Some menu text may contain display-name junk;
                 * still accept an exact @handle occurrence.
                 */
                if (
                    lower.includes(
                        '@' +
                        normalizedHandle
                    )
                ) {

                    return REL.FOLLOWING;
                }
            }


            /*
             * Examples:
             *
             *     Follow @Reuters
             *     Follow Reuters
             *     Follow
             */
            if (
                lower === 'follow'
            ) {

                return REL.NOT_FOLLOWING;
            }


            if (
                lower.startsWith(
                    'follow '
                )
            ) {

                const possibleHandle =
                    lower
                        .slice(
                            'follow '.length
                        )
                        .replace(/^@/, '')
                        .trim();


                if (
                    !possibleHandle ||
                    possibleHandle ===
                        normalizedHandle
                ) {

                    return REL.NOT_FOLLOWING;
                }


                if (
                    lower.includes(
                        '@' +
                        normalizedHandle
                    )
                ) {

                    return REL.NOT_FOLLOWING;
                }
            }
        }


        return REL.UNKNOWN;
    }


    // ============================================================
    // WAIT FOR NEW MORE MENU
    // ============================================================

    function waitForMenu(
        menusBefore
    ) {

        return new Promise(
            resolve => {

                const started =
                    performance.now();


                function check() {

                    const menus =
                        getVisibleMenus();


                    /*
                     * Prefer a menu that did not exist before
                     * our click.
                     */
                    const newMenu =
                        menus.find(
                            menu =>
                                !menusBefore.has(
                                    menu
                                )
                        );


                    if (newMenu) {

                        resolve(
                            newMenu
                        );

                        return;
                    }


                    /*
                     * Sometimes X recycles the same portal element.
                     *
                     * If there was no menu before and exactly one
                     * exists now, that's ours.
                     */
                    if (
                        menusBefore.size === 0 &&
                        menus.length === 1
                    ) {

                        resolve(
                            menus[0]
                        );

                        return;
                    }


                    if (
                        performance.now() -
                        started >
                        MENU_TIMEOUT
                    ) {

                        resolve(null);

                        return;
                    }


                    requestAnimationFrame(
                        check
                    );
                }


                check();
            }
        );
    }


    // ============================================================
    // CLOSE MENU
    // ============================================================

    async function closeMenu(
        menu,
        caret
    ) {

        if (!menu) {
            return;
        }


        /*
         * X responds to Escape for menus.
         */
        const event =
            new KeyboardEvent(
                'keydown',
                {
                    key: 'Escape',
                    code: 'Escape',
                    keyCode: 27,
                    which: 27,
                    bubbles: true,
                    cancelable: true
                }
            );


        document.dispatchEvent(
            event
        );


        await sleep(75);


        /*
         * Fallback: toggle the caret again.
         */
        if (
            menu.isConnected &&
            isVisible(menu) &&
            caret?.isConnected
        ) {

            caret.click();

            await sleep(75);
        }
    }


    // ============================================================
    // CHECK ONE ACCOUNT USING ITS TWEET MENU
    // ============================================================

    async function inspectTweetRelationship(
        tweet,
        handle
    ) {

        if (
            !(tweet instanceof Element) ||
            !tweet.isConnected
        ) {

            return REL.UNKNOWN;
        }


        /*
         * Don't stomp on a menu the user is currently using.
         */
        if (
            userMenuIsOpen()
        ) {

            return REL.BUSY;
        }


        const caret =
            getTweetCaret(
                tweet
            );


        if (!caret) {

            log(
                `⚠ No More button for @${handle}`
            );

            return REL.UNKNOWN;
        }


        const existingMenus =
            new Set(
                getVisibleMenus()
            );


        /*
         * Programmatically open this tweet's More menu.
         */
        caret.click();


        const menu =
            await waitForMenu(
                existingMenus
            );


        if (!menu) {

            log(
                `⚠ More menu did not open for @${handle}`
            );

            return REL.UNKNOWN;
        }


        const texts =
            getMenuItemTexts(
                menu
            );


        const relationship =
            classifyMenu(
                texts,
                handle
            );


        if (DEBUG) {

            log(
                `Menu for @${handle}:`,
                texts
            );
        }


        await closeMenu(
            menu,
            caret
        );


        return relationship;
    }


    // ============================================================
    // FIND CURRENT TWEET FOR HANDLE
    // ============================================================

    function findTweetByHandle(handle) {

        const normalized =
            handle.toLowerCase();


        for (
            const tweet
            of document.querySelectorAll(
                'article[data-testid="tweet"]'
            )
        ) {

            const found =
                getTweetHandle(
                    tweet
                );


            if (
                found &&
                found.toLowerCase() ===
                    normalized
            ) {

                return tweet;
            }
        }


        return null;
    }


    // ============================================================
    // VISIBILITY
    // ============================================================

    function hideTweet(
        tweet,
        handle
    ) {

        if (
            tweet.getAttribute(
                HIDDEN_ATTR
            ) === '1'
        ) {
            return;
        }


        tweet.setAttribute(
            HIDDEN_ATTR,
            '1'
        );


        blockedCount++;


        log(
            `🚫 Hidden @${handle} — X menu says Follow`,
            {
                blockedCount
            }
        );
    }


    function showTweet(tweet) {

        tweet?.removeAttribute(
            HIDDEN_ATTR
        );
    }


    function applyRelationship(
        handle,
        relationship
    ) {

        const normalized =
            handle.toLowerCase();


        for (
            const tweet
            of document.querySelectorAll(
                'article[data-testid="tweet"]'
            )
        ) {

            const tweetHandle =
                getTweetHandle(
                    tweet
                );


            if (
                !tweetHandle ||
                tweetHandle.toLowerCase() !==
                    normalized
            ) {
                continue;
            }


            if (
                relationship ===
                REL.NOT_FOLLOWING
            ) {

                hideTweet(
                    tweet,
                    tweetHandle
                );

            } else {

                /*
                 * FOLLOWING or UNKNOWN:
                 * fail open and show it.
                 */
                showTweet(
                    tweet
                );
            }
        }
    }


    // ============================================================
    // QUEUE
    // ============================================================

    function queueHandle(
        handle
    ) {

        const key =
            handle.toLowerCase();


        if (
            getCachedRelationship(
                handle
            )
        ) {
            return;
        }


        if (
            queuedHandles.has(
                key
            )
        ) {
            return;
        }


        queuedHandles.add(
            key
        );


        checkQueue.push(
            handle
        );


        log(
            `🔎 Queued @${handle}`
        );


        runWorker();
    }


    async function runWorker() {

        if (workerRunning) {
            return;
        }


        workerRunning =
            true;


        try {

            while (
                checkQueue.length > 0
            ) {

                const handle =
                    checkQueue.shift();


                const key =
                    handle.toLowerCase();


                const cached =
                    getCachedRelationship(
                        handle
                    );


                if (cached) {

                    queuedHandles.delete(
                        key
                    );

                    applyRelationship(
                        handle,
                        cached
                    );

                    continue;
                }


                /*
                 * X may recycle/remove the exact tweet that
                 * originally queued the handle.
                 *
                 * Find another current tweet from that author.
                 */
                const tweet =
                    findTweetByHandle(
                        handle
                    );


                if (!tweet) {

                    /*
                     * No longer visible.
                     * Do not cache UNKNOWN.
                     */
                    queuedHandles.delete(
                        key
                    );

                    continue;
                }


                log(
                    `🔍 Checking @${handle} via tweet menu...`
                );


                const relationship =
                    await inspectTweetRelationship(
                        tweet,
                        handle
                    );


                /*
                 * User happened to have a menu open.
                 * Requeue instead of interfering.
                 */
                if (
                    relationship ===
                    REL.BUSY
                ) {

                    checkQueue.push(
                        handle
                    );

                    await sleep(
                        1000
                    );

                    continue;
                }


                cacheRelationship(
                    handle,
                    relationship
                );


                queuedHandles.delete(
                    key
                );


                if (
                    relationship ===
                    REL.NOT_FOLLOWING
                ) {

                    log(
                        `🚫 @${handle}: NOT FOLLOWING`
                    );

                } else if (
                    relationship ===
                    REL.FOLLOWING
                ) {

                    log(
                        `✓ @${handle}: FOLLOWING`
                    );

                } else {

                    log(
                        `? @${handle}: relationship unknown — keeping`
                    );
                }


                applyRelationship(
                    handle,
                    relationship
                );


                await sleep(
                    CHECK_DELAY
                );
            }

        } finally {

            workerRunning =
                false;
        }
    }


    // ============================================================
    // PROCESS TWEET
    // ============================================================

    function processTweet(tweet) {

        if (
            !(tweet instanceof Element)
        ) {
            return;
        }


        if (!isHome()) {

            showTweet(
                tweet
            );

            return;
        }


        const handle =
            getTweetHandle(
                tweet
            );


        if (!handle) {

            showTweet(
                tweet
            );

            return;
        }


        /*
         * Protect against X timeline virtualization/recycling.
         */
        const previousHandle =
            tweet.getAttribute(
                HANDLE_ATTR
            );


        if (
            previousHandle &&
            previousHandle.toLowerCase() !==
                handle.toLowerCase()
        ) {

            showTweet(
                tweet
            );
        }


        tweet.setAttribute(
            HANDLE_ATTR,
            handle
        );


        const relationship =
            getCachedRelationship(
                handle
            );


        if (
            relationship ===
            REL.NOT_FOLLOWING
        ) {

            hideTweet(
                tweet,
                handle
            );

            return;
        }


        /*
         * FOLLOWING and cached UNKNOWN stay visible.
         */
        if (relationship) {

            showTweet(
                tweet
            );

            return;
        }


        /*
         * Relationship unknown.
         *
         * Leave visible until X positively tells us Follow.
         */
        showTweet(
            tweet
        );


        queueHandle(
            handle
        );
    }


    // ============================================================
    // SCAN
    // ============================================================

    function scan(root) {

        if (!(root instanceof Element)) {
            return;
        }


        if (
            root.matches?.(
                'article[data-testid="tweet"]'
            )
        ) {

            processTweet(
                root
            );
        }


        root
            .querySelectorAll(
                'article[data-testid="tweet"]'
            )
            .forEach(
                processTweet
            );
    }


    function queueScan(root) {

        if (!(root instanceof Element)) {
            return;
        }


        pendingRoots.add(
            root
        );


        if (scanScheduled) {
            return;
        }


        scanScheduled =
            true;


        requestAnimationFrame(
            () => {

                scanScheduled =
                    false;


                const roots =
                    [...pendingRoots];


                pendingRoots.clear();


                for (
                    const root
                    of roots
                ) {

                    if (
                        root.isConnected
                    ) {

                        scan(
                            root
                        );
                    }
                }
            }
        );
    }


    // ============================================================
    // OBSERVER
    // ============================================================

    const observer =
        new MutationObserver(
            mutations => {

                for (
                    const mutation
                    of mutations
                ) {

                    if (
                        mutation.type !==
                        'childList'
                    ) {
                        continue;
                    }


                    /*
                     * X frequently recycles an existing article.
                     */
                    if (
                        mutation.target
                        instanceof Element
                    ) {

                        const tweet =
                            mutation.target.closest(
                                'article[data-testid="tweet"]'
                            );


                        if (tweet) {

                            queueScan(
                                tweet
                            );
                        }
                    }


                    for (
                        const node
                        of mutation.addedNodes
                    ) {

                        if (
                            node instanceof Element
                        ) {

                            queueScan(
                                node
                            );
                        }
                    }
                }
            }
        );


    // ============================================================
    // DEBUG API
    // ============================================================

    window.XFollowFilter = {

        stats() {

            const values =
                Object.values(
                    cache
                );


            const result = {

                cached:
                    values.length,

                following:
                    values.filter(
                        x =>
                            x.relationship ===
                            REL.FOLLOWING
                    ).length,

                notFollowing:
                    values.filter(
                        x =>
                            x.relationship ===
                            REL.NOT_FOLLOWING
                    ).length,

                unknown:
                    values.filter(
                        x =>
                            x.relationship ===
                            REL.UNKNOWN
                    ).length,

                queued:
                    checkQueue.length,

                blockedCount,

                workerRunning
            };


            console.table(
                result
            );


            return result;
        },


        getCache() {

            console.table(
                Object.values(
                    cache
                )
            );


            return cache;
        },


        clearCache() {

            cache = {};

            localStorage.removeItem(
                CACHE_KEY
            );


            document
                .querySelectorAll(
                    `article[${HIDDEN_ATTR}]`
                )
                .forEach(
                    showTweet
                );


            queuedHandles.clear();

            checkQueue.length = 0;


            log(
                'Cache cleared.'
            );


            scan(
                document.documentElement
            );
        },


        recheck(handle) {

            const clean =
                String(handle || '')
                    .replace(/^@/, '')
                    .trim();


            if (!clean) {
                return;
            }


            delete cache[
                clean.toLowerCase()
            ];


            saveCache();


            queuedHandles.delete(
                clean.toLowerCase()
            );


            queueHandle(
                clean
            );
        }
    };


    // ============================================================
    // GO
    // ============================================================

    installCSS();


    scan(
        document.documentElement
    );


    observer.observe(
        document.documentElement,
        {
            subtree: true,
            childList: true
        }
    );


    log(
        'Observer active.'
    );

})();
```
